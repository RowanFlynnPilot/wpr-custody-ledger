--- JavaScript tab ---

window.addEventListener("message", function (e) {
  if (!e.data || e.data.source !== "wpr-custody-ledger") return;
  var frame = document.getElementById("custody-ledger");
  if (!frame || e.origin !== new URL(frame.src).origin) return;
  if (typeof e.data.height === "number" && e.data.height > 0) {
    frame.style.height = Math.ceil(e.data.height) + "px";
  }
  // A jump link inside the tool: the frame is full height and cannot scroll itself, so scroll the page.
  if (typeof e.data.scrollTo === "number") {
    window.scrollTo({ top: frame.getBoundingClientRect().top + window.scrollY + e.data.scrollTo - 16, behavior: "smooth" });
  }
});
// Optional: lets a shared link such as  your-article-url#facility=lincoln-hills  open that facility.
(function () {
  var f = document.getElementById("custody-ledger");
  if (f && /^#facility=[a-z0-9-]+$/.test(location.hash)) f.src = f.src.split("#")[0] + location.hash;
})();
